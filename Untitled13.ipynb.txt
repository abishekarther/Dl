        "import tensorflow as tf\n",
        "import numpy as np\n",
        "\n",
        "# Small parallel dataset\n",
        "eng = [\"hello\", \"how are you\", \"good morning\", \"thank you\"]\n",
        "fra = [\"bonjour\", \"comment allez vous\", \"bonjour\", \"merci\"]\n",
        "\n",
        "# Tokenization\n",
        "en_tok = tf.keras.preprocessing.text.Tokenizer()\n",
        "fr_tok = tf.keras.preprocessing.text.Tokenizer()\n",
        "\n",
        "en_tok.fit_on_texts(eng)\n",
        "fr_tok.fit_on_texts(fra)\n",
        "\n",
        "X = en_tok.texts_to_sequences(eng)\n",
        "Y = fr_tok.texts_to_sequences(fra)\n",
        "\n",
        "X = tf.keras.preprocessing.sequence.pad_sequences(X, padding='post')\n",
        "Y = tf.keras.preprocessing.sequence.pad_sequences(Y, padding='post')\n",
        "\n",
        "# Encoder\n",
        "encoder_input = tf.keras.Input(shape=(None,))\n",
        "enc_emb = tf.keras.layers.Embedding(len(en_tok.word_index)+1, 32)(encoder_input)\n",
        "_, state_h, state_c = tf.keras.layers.LSTM(\n",
        "    32, return_state=True\n",
        ")(enc_emb)\n",
        "\n",
        "# Decoder\n",
        "decoder_input = tf.keras.Input(shape=(None,))\n",
        "dec_emb = tf.keras.layers.Embedding(len(fr_tok.word_index)+1, 32)(decoder_input)\n",
        "dec_out = tf.keras.layers.LSTM(32, return_sequences=True)(\n",
        "    dec_emb, initial_state=[state_h, state_c]\n",
        ")\n",
        "output = tf.keras.layers.Dense(\n",
        "    len(fr_tok.word_index)+1, activation='softmax'\n",
        ")(dec_out)\n",
        "\n",
        "# Build and train\n",
        "model = tf.keras.Model([encoder_input, decoder_input], output)\n",
        "model.compile(optimizer='adam', loss='sparse_categorical_crossentropy')\n",
        "\n",
        "model.fit([X, Y], np.expand_dims(Y, -1), epochs=100, verbose=0)\n",
        "\n",
        "print(\"Encoder-Decoder model trained successfully!\")"
